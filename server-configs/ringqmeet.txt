server {
    listen 8445 ssl http2;
	listen [::]:8445 ssl http2;

	server_name ringq;

	ssl_certificate /etc/letsencrypt/live/ssl.ringq.ai/fullchain.pem;
	ssl_certificate_key /etc/letsencrypt/live/ssl.ringq.ai/privkey.pem;
	ssl_protocols	        TLSv1 TLSv1.1 TLSv1.2 TLSv1.3;
	ssl_prefer_server_ciphers on;
	ssl_ciphers             ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384:ECDHE-ECDSA-CHACHA20-POLY1305:ECDHE-RSA-CHACHA20-POLY1305:DHE-RSA-AES128-GCM-SHA256:DHE-RSA-AES256-GCM-SHA384:DHE-RSA-AES256-SHA:AES256-SHA;
	ssl_session_cache       shared:SSL:40m;
	ssl_session_timeout     2h;
	ssl_session_tickets     off;

    root /var/www/ringqmeet/public;
    index index.php index.html;

    add_header Access-Control-Allow-Origin $http_origin always;
    add_header Access-Control-Allow-Credentials true always;
    add_header Access-Control-Allow-Headers 'authorization, access-control-allow-origin, cookie, content-type';
    #add_header Access-Control-Expose-Headers 'Authorization' always;

    sendfile on;
    client_max_body_size    100m;
    client_body_buffer_size 128k;

    fastcgi_connect_timeout 600;
    fastcgi_send_timeout 600;
    fastcgi_read_timeout 300;

    fastcgi_buffering on;
    fastcgi_buffer_size 128k;
    fastcgi_buffers 4 256k;
    fastcgi_busy_buffers_size 256k;
    #fastcgi_max_temp_file_size 2048m;
    #fastcgi_temp_file_write_size 2048m;
    #fastcgi_temp_path /etc/nginx/proxy_temp;

    fastcgi_keep_conn on;
    fastcgi_socket_keepalive on;

    location ~* \.(jpg|jpeg|gif|png|css|js|ico|xml)$ {
        access_log        off;
        log_not_found     off;
        expires           10d;
    }

    location ~* \.(?:ico|cur|gz|svg|svgz|mp4|ogg|ogv|webm|htc|woff)$ {
        expires 1M;
        access_log off;
        log_not_found off;
        add_header Cache-Control "public";
    }

    location ~* \.(?:css|js)$ {
        gzip_vary on;
        expires 1M;
        access_log off;
        log_not_found off;
        add_header Cache-Control "public";
    }

    location / {
        try_files $uri $uri/ /index.php?_url=$uri&$args;
    }

    location ~ [^/]\.php(/|$) {
        fastcgi_pass socket;
        fastcgi_index /index.php;

        include fastcgi_params;
        fastcgi_split_path_info ^(.+?\.php)(/.*)$;
        if (!-f $document_root$fastcgi_script_name) {
            return 404;
        }
        
        fastcgi_param PATH_INFO       $fastcgi_path_info;
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
    }

    proxy_http_version 1.1;
    proxy_set_header Connection "";
    proxy_set_header HTTPS "on";
    proxy_set_header X-Real-IP  $remote_addr;
    proxy_set_header X-Forwarded-For $remote_addr;
    proxy_set_header Host $host;
    proxy_set_header Upgrade $http_upgrade;
    proxy_set_header Connection $connection_upgrade;
    proxy_connect_timeout 45;
    proxy_send_timeout 45;
    proxy_read_timeout 36000s;
    proxy_buffering off;
    proxy_buffer_size 4k;
    proxy_buffers 4 32k;
    proxy_busy_buffers_size 64k;
    #proxy_max_temp_file_size 2048m;
    #proxy_temp_file_write_size 64k;
    #proxy_temp_path /etc/nginx/proxy_temp;
    
    location /img { location ~ \.php$ {return 403;} }	
    location /avr { location ~ \.php$ {return 403;} }
    location /fbk { location ~ \.php$ {return 403;} }
    location /temp { location ~ \.php$ {return 403;} }

    location = /favicon.ico { access_log off; log_not_found off; }
    location = /robots.txt { access_log off; log_not_found off; }
    location ~ /\.ht { deny all; access_log off; log_not_found off; }
}
